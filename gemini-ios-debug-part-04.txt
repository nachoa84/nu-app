    (() => {
      try {
        const state = JSON.parse(localStorage.getItem("routineState") || "null");
        const day = Number(state?.currentDay || localStorage.getItem("selectedDay") || 1);
        if (!Number.isFinite(day) || day < 1) return;

        const routineDay = document.getElementById("homeRoutineDay");
        const routinePercent = document.getElementById("homeRoutinePercent");
        const routineFill = document.getElementById("homeRoutineProgressFill");
        const routineContinue = document.getElementById("homeRoutineContinueLabel");
        const chatTitle = document.querySelector(".chat-header strong");
        const percent = Math.min(100, Math.round((day / 30) * 100));

        if (routineDay) routineDay.textContent = `Día ${day} de 30`;
        if (routinePercent) routinePercent.textContent = `${percent}%`;
        if (routineFill) routineFill.style.width = `${percent}%`;
        if (routineContinue) routineContinue.textContent = `Continuar Día ${day}`;
        if (chatTitle) chatTitle.textContent = `Día ${day} · Rutina 30 Días`;
      } catch (_) {}
    })();
  </script>
